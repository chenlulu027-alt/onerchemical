export function getArticle(slug: string): BlogArticle | undefined {
  return ARTICLES.find((a) => a.slug === slug && !a.draft);
}

export function articlesByCluster(cluster: ClusterKey): BlogArticle[] {
  return ARTICLES.filter((a) => a.cluster === cluster && !a.draft);
}

export function publishedArticles(): BlogArticle[] {
  return ARTICLES.filter((a) => !a.draft).sort((a, b) =>
    b.datePublished.localeCompare(a.datePublished),
  );
}

'use client';

import { useMemo, useState } from 'react';
import { AnimatePresence, motion } from 'motion/react';
import { LocalizedLink } from '@/components/LocalizedLink';
import { CLUSTERS, type BlogArticle, type ClusterKey } from '@/lib/blog-registry';

type Filter = 'all' | ClusterKey;

export function BlogListClient({ articles }: { articles: BlogArticle[] }) {
  const [filter, setFilter] = useState<Filter>('all');
  const filtered = useMemo(
    () => (filter === 'all' ? articles : articles.filter((a) => a.cluster === filter)),
    [filter, articles],
  );
  const clusterKeys = Object.keys(CLUSTERS) as ClusterKey[];

  return (
    <section className="mx-auto max-w-5xl px-6 py-16">
      <header className="mb-10 border-b border-ink-100 pb-10">
        <h1 className="mb-3 font-serif text-5xl font-normal text-ink-900">Research</h1>
        <p className="max-w-2xl text-lg text-ink-500">
          Technical articles on cosmetic preservation chemistry, challenge testing, regulation, and
          formulation strategy.
        </p>
      </header>

      <nav aria-label="Cluster filter" className="relative mb-10 flex flex-wrap gap-1.5">
        <Tab active={filter === 'all'} id="all" onClick={() => setFilter('all')}>All</Tab>
        {clusterKeys.map((k) => (
          <Tab key={k} active={filter === k} id={k} onClick={() => setFilter(k)}>
            {CLUSTERS[k].label}
          </Tab>
        ))}
      </nav>

      {filtered.length === 0 ? (
        <p className="text-ink-500">No articles in this cluster yet.</p>
      ) : (
        <ul className="divide-y divide-ink-100 border-t border-ink-100">
          <AnimatePresence mode="popLayout" initial={false}>
            {filtered.map((a, i) => (
              <motion.li
                key={a.slug}
                layout
                initial={{ opacity: 0, y: 10 }}
                animate={{ opacity: 1, y: 0 }}
                exit={{ opacity: 0, y: -6 }}
                transition={{ duration: 0.3, delay: Math.min(i * 0.04, 0.2) }}
              >
                <LocalizedLink
                  href={`/blog/${a.slug}`}
                  className="group block py-6 transition hover:bg-paper/60"

#!/usr/bin/env node
/**
 * Build /public/search-index.json from src/lib/blog-registry.ts.
 * Runs as a `prebuild` hook 鈥?zero runtime cost, pure static JSON.
 *
 * We don't import blog-registry at build time (it's TS). We parse it
 * as plain text with a loose regex 鈥?the registry file is single-purpose
 * and the shape is stable. If the shape changes, update this script.
 *
 * Design choice: no AST parser dependency. The parse is lenient 鈥?a
 * missing field throws, and the build fails fast with a clear message.
 */

import { readFileSync, writeFileSync, mkdirSync } from 'node:fs';
import { dirname, resolve } from 'node:path';
import { fileURLToPath } from 'node:url';

const __dirname = dirname(fileURLToPath(import.meta.url));
const ROOT = resolve(__dirname, '..');
const REGISTRY = resolve(ROOT, 'src/lib/blog-registry.ts');
const OUT = resolve(ROOT, 'public/search-index.json');

const src = readFileSync(REGISTRY, 'utf8');

/* --------- extract CLUSTERS { key: { label, description } } -------------- */
const clustersMatch = src.match(/export const CLUSTERS\s*=\s*({[\s\S]*?})\s*as\s+const;/);
if (!clustersMatch) throw new Error('build-search-index: CLUSTERS block not found');
const clustersRaw = clustersMatch[1];
const clusterLabels = {};
for (const m of clustersRaw.matchAll(/['"]([a-z0-9-]+)['"]\s*:\s*{[^}]*?label\s*:\s*['"]([^'"]+)['"]/g)) {
  clusterLabels[m[1]] = m[2];
}

/* --------- extract ARTICLES array -------------------------------------- */
const articlesMatch = src.match(/export const ARTICLES\s*:\s*BlogArticle\[\]\s*=\s*(\[[\s\S]*?\]);/);
if (!articlesMatch) throw new Error('build-search-index: ARTICLES array not found');

// Simple field extractor: find each {...} object in the array
const objRE = /{[^{}]*(?:{[^{}]*}[^{}]*)*}/g;
const objects = articlesMatch[1].match(objRE) ?? [];

function pick(obj, key) {
  const m = obj.match(new RegExp('\\b' + key + '\\s*:\\s*([\'"])((?:[^\\\\]|\\\\.)*?)\\1'));
  return m ? m[2].replace(/\\(['"\\])/g, '$1') : undefined;
}
function pickBool(obj, key) {
  const m = obj.match(new RegExp('\\b' + key + '\\s*:\\s*(true|false)'));
  return m ? m[1] === 'true' : false;
}
function pickArr(obj, key) {
  const m = obj.match(new RegExp('\\b' + key + '\\s*:\\s*(\\[[^\\]]*\\])'));
  if (!m) return [];
  return [...m[1].matchAll(/['"]([^'"]+)['"]/g)].map((x) => x[1]);
}

const docs = [];
for (const o of objects) {
  if (pickBool(o, 'draft')) continue;
  const slug = pick(o, 'slug');
  const title = pick(o, 'title');
  const excerpt = pick(o, 'excerpt');
  const cluster = pick(o, 'cluster');
  const searchKeywords = pickArr(o, 'searchKeywords');
  if (!slug || !title || !excerpt || !cluster) {
    throw new Error('build-search-index: missing required field in ' + JSON.stringify(o).slice(0, 120));
  }
  docs.push({
    slug,
    title,
    excerpt,
    cluster: clusterLabels[cluster] ?? cluster,
    searchKeywords,
    type: 'article',
    url: `/blog/${slug}`,
  });
}

/* --------- add static pages worth searching ---------------------------- */
docs.push(
  { slug: 'about', title: 'About Onerchemical', excerpt: 'The publication, who writes, what we don\u2019t do.', cluster: 'About', searchKeywords: ['about', 'company', 'publisher'], type: 'page', url: '/about' },
  { slug: 'yang-ruosong', title: 'Yang Ruosong', excerpt: 'Head of Technical Content 鈥?cosmetic preservation articles.', cluster: 'Author', searchKeywords: ['yang ruosong', 'author', 'bio'], type: 'page', url: '/yang-ruosong' },
);

mkdirSync(dirname(OUT), { recursive: true });
writeFileSync(OUT, JSON.stringify(docs));
console.log(`[search-index] wrote ${docs.length} docs 鈫?public/search-index.json`);

type Entry = MetadataRoute.Sitemap[number];

/**
 * makeEntry wraps a canonical path + optional lastModified into a sitemap
 * entry with hreflang alternates baked in. Phase 0: EN only, but the
 * alternates field already carries x-default so Google can start reading it.
 */
function makeEntry(
  path: string,
  opts: {
    lastModified?: string | Date;
    changeFrequency?: Entry['changeFrequency'];
    priority?: number;
  } = {},
): Entry {
  return {
    url: `${SITE.url}${path}`,
    lastModified: opts.lastModified ? new Date(opts.lastModified) : new Date(),
    changeFrequency: opts.changeFrequency ?? 'monthly',
    priority: opts.priority ?? 0.5,
    alternates: {
      languages: makeAlternates(path, SITE.url),
    },
  };
}

export default function sitemap(): MetadataRoute.Sitemap {
  const staticEntries: Entry[] = [
    makeEntry('/', { changeFrequency: 'weekly', priority: 1.0 }),
    makeEntry('/blog', { changeFrequency: 'weekly', priority: 0.8 }),
    makeEntry('/about', { changeFrequency: 'yearly', priority: 0.4 }),
    makeEntry('/yang-ruosong', { changeFrequency: 'monthly', priority: 0.5 }),
  ];

  const articleEntries: Entry[] = publishedArticles().map((a) =>
    makeEntry(`/blog/${a.slug}`, {
      lastModified: a.dateModified ?? a.datePublished,
      changeFrequency: 'yearly',
      priority: 0.7,
    }),
  );

  const productEntries: Entry[] = PRODUCTS.map((p) =>
    makeEntry(`/products/${p.slug}`, {
      changeFrequency: 'monthly',
      priority: 0.6,
    }),
  );
